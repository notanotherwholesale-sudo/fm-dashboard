    (function(){
      function nawRate(){
        var base = {{ product.title | json }}.replace(/\s*x\d+\s*$/i,'').trim().toLowerCase();
        fetch("{{ 'naw-reviews-data.json' | asset_url }}")
          .then(function(r){return r.json();})
          .then(function(rv){
            var m = rv.filter(function(x){ return x.p && String(x.p).trim().toLowerCase()===base && (+x.s>0); });
            if(m.length < 1) return;
            var avg = m.reduce(function(s,x){return s+(+x.s||0);},0)/m.length;
            document.querySelectorAll('script[type="application/ld+json"]').forEach(function(sc){
              try{
                var j=JSON.parse(sc.textContent);
                if(j['@type']==='Product' && !j.aggregateRating){
                  j.aggregateRating={"@type":"AggregateRating","ratingValue":avg.toFixed(1),"reviewCount":m.length,"bestRating":5,"worstRating":1};
                  sc.textContent=JSON.stringify(j);
                }
              }catch(e){}
            });
          }).catch(function(){});
      }
      if(document.readyState==='loading'){document.addEventListener('DOMContentLoaded',nawRate);}else{nawRate();}
    })();

      NAW mobile viewport lock — two fixes for the "loose" page on phones:
      1. overflow-x clamp: any element that momentarily renders wider than the
         screen (sliders, marquees, app embeds) un-anchors the whole page on iOS.
         overflow:clip preferred (doesn't break position:sticky); hidden fallback.
      2. 16px minimum on form fields at mobile widths: iOS Safari force-zooms the
         page when a field under 16px is focused (search 12px, footer email 13px,
         cart qty 10.8px all triggered it) and the page stays zoomed after.
    {%- endcomment -%}

(function () {
  const KEY = "__dispatchTimerInterval";

  function startDispatchTimer() {
    const textEl = document.getElementById("dispatchText");
    if (!textEl) return;

    // Prevent multiple timers if theme re-renders / loads twice
    if (window[KEY]) clearInterval(window[KEY]);

    const CUTOFF_HOUR = 11;

    const isMonFri = d => d.getDay() >= 1 && d.getDay() <= 5;
    const isSunThu = d => d.getDay() === 0 || (d.getDay() >= 1 && d.getDay() <= 4);

    const setTime = (d, h, m = 0) => {
      const x = new Date(d);
      x.setHours(h, m, 0, 0);
      return x;
    };

    const addDays = (d, days) => {
      const x = new Date(d);
      x.setDate(x.getDate() + days);
      return x;
    };

    const ordinal = n => {
      const s = ["th","st","nd","rd"];
      const v = n % 100;
      return n + (s[(v - 20) % 10] || s[v] || s[0]);
    };

    const weekdayName = d =>
      new Intl.DateTimeFormat(undefined, { weekday: "long" }).format(d);

    const nextMonday = now => {
      const day = now.getDay();
      const daysUntil = (8 - day) % 7 || 7;
      return addDays(now, daysUntil);
    };

    // Helpers
    const setText = (str) => { textEl.textContent = str; };
    const setDispatchHtml = (dayLabel, dateNum) => {
      textEl.innerHTML = `Order now for dispatch <span class="dispatch-date">${dayLabel} ${dateNum}</span>`;
    };

    function render() {
      const now = new Date();
      const day = now.getDay();
      const hour = now.getHours();
      const cutoff = setTime(now, CUTOFF_HOUR);

      const isFriday = day === 5;
      const isSaturday = day === 6;
      const isSunday = day === 0;

      // STATE 4 — Fri 11:00 → Sun 10:59 (dispatch Monday)
      if ((isFriday && now >= cutoff) || isSaturday || (isSunday && hour < 11)) {
        const monday = nextMonday(now);
        setDispatchHtml("Monday", ordinal(monday.getDate()));
        return;
      }

      // STATE 1 — Sun–Thu 11:00–23:59 (dispatch next day)
      if (isSunThu(now) && now >= cutoff) {
        const dispatch = addDays(now, 1);
        setDispatchHtml(weekdayName(dispatch), ordinal(dispatch.getDate()));
        return;
      }

      // STATE 2 — Mon–Fri 00:00–07:59 (hours only)
      if (isMonFri(now) && hour < 8) {
        const msLeft = cutoff - now;
        const hours = Math.max(1, Math.ceil(msLeft / 3600000));
        setText(`Order within ${hours} hour${hours === 1 ? "" : "s"} and get same day dispatch`);
        return;
      }

      // STATE 3 — Mon–Fri 08:00–10:59 (live timer)
      if (isMonFri(now) && now < cutoff && hour >= 8) {
        const msLeft = cutoff - now;
        const totalMinutes = Math.floor(msLeft / 60000);
        const h = Math.floor(totalMinutes / 60);
        const m = totalMinutes % 60;

        if (h > 0) {
          setText(`Order within ${h} hour${h === 1 ? "" : "s"}: ${String(m).padStart(2,"0")} minutes and get same day dispatch`);
        } else {
          setText(`Order within ${m} minute${m === 1 ? "" : "s"} and get same day dispatch`);
        }
        return;
      }

      // Fallback (dispatch tomorrow)
      const tomorrow = addDays(now, 1);
      setDispatchHtml(weekdayName(tomorrow), ordinal(tomorrow.getDate()));
    }

    render();
    window[KEY] = setInterval(render, 1000);
  }

  // Run on initial load
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", startDispatchTimer);
  } else {
    startDispatchTimer();
  }

  // Re-run when Shopify sections re-render
  document.addEventListener("shopify:section:load", startDispatchTimer);
})();
</script>
